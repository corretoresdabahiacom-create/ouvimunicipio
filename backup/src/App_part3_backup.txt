                          Registrar Pasta

                                <span className="font-bold">Responsável: <span className="font-normal text-stone-600">{s.responsavel}</span></span>
                                <span className="font-bold font-mono">Mail: <span className="font-normal text-[#1351B4]">{s.email}</span></span>

                    <div className="pb-5 border-b border-stone-100 mb-6">

                          >
                            <option value="">Selecione a secretaria...</option>
                            {secretarias.map(s => (
                              <option key={s.id} value={s.nome}>{s.nome}</option>
                            ))}
                          </select>
                        </div>

                        <div>
                          <label htmlFor="rep-servidor-select" className="block text-[10px] font-bold text-stone-500 mb-1">ENCAMINHAR DILIGÊNCIA PARA O ASSISTENTE / SERVIDOR *</label>
                          <select
                            id="rep-servidor-select"
                            value={diligenciadoName}
                            onChange={(e) => setDiligenciadoName(e.target.value)}

                          >
                            <option value="">Selecione o servidor...</option>
                            {usuarios
                              .filter(u => !reportSecretaria || u.secretaria?.toLowerCase().trim() === reportSecretaria.toLowerCase().trim() || u.perfil === "Ouvidoria")
                              .map(u => (

                              ))
                            }
                            <option value="Outro">Outro encarregado (Digitar nome)</option>
                          </select>
                        </div>

                        {diligenciadoName === "Outro" && (
                          <div className="animate-fade-in">
                            <label htmlFor="rep-custom-servidor" className="block text-[10px] font-bold text-stone-500 mb-1">NOME DO PROFISSIONAL DESIGNADO *</label>
                            <input
                              id="rep-custom-servidor"
                              type="text"
                              value={customDiligenciadoName}
                              onChange={(e) => setCustomDiligenciadoName(e.target.value)}
                              className="w-full text-xs px-3 py-2 border rounded-lg bg-white outline-none active:ring-1"
                              placeholder="Ex: Pedro de Alcantara"
                            />
                          </div>

                        <div className="flex gap-2.5 mt-2">
                          <button

                              const filteredReportManifs = manifestacoes.filter(m => m.secretaria?.toLowerCase().trim() === activeSec?.toLowerCase().trim());
                              const reportTotal = filteredReportManifs.length;
                              const reportLidas = filteredReportManifs.filter(m => m.status !== "Pendente").length;
                              const reportResolvidas = filteredReportManifs.filter(m => m.status === "Resolvido").length;
                              const reportPendentes = filteredReportManifs.filter(m => m.status === "Pendente" || m.status === "Urgente").length;
                              const finalDiligencia = (diligenciadoName && diligenciadoName !== "Outro") ? diligenciadoName : (customDiligenciadoName || "Não designado");

========================================================
Secretaria: ${activeSec}
Emissão: ${new Date().toLocaleDateString('pt-BR')}
Diligenciado para: ${finalDiligencia}

HISTÓRICO OPERACIONAL (MÊS DE MAIO):
• Quantidade de Mensagens recebidas: ${reportTotal}
• Quantidade de Mensagens Lidas/Triadas: ${reportLidas}
• Quantidade de Mensagens Resolvidas: ${reportResolvidas}
• Quantidade de Mensagens Pendentes: ${reportPendentes}

DILIGÊNCIA ENCAMINHADA COM SUCESSO.
REGIMENTO DE COOPERAÇÃO TÉCNICA LAI Nº 12.527/2011.
========================================================`;

                              link.download = `relatorio-${activeSec.toLowerCase().replace(/\s+/g, "")}-${Date.now()}.txt`;
                              document.body.appendChild(link);
                              link.click();
                              document.body.removeChild(link);
                              URL.revokeObjectURL(url);
                            }}
                            className="bg-[#071D41] hover:bg-slate-800 disabled:bg-stone-200 disabled:text-stone-450 disabled:cursor-not-allowed text-[#FFCD07] font-extrabold text-xs py-3 px-4 rounded-xl transition-all shadow-sm flex items-center justify-center gap-1.5 cursor-pointer"
                          >
                            <Download className="w-4 h-4" /> Exportar .TXT (LAI)
                          </button>
                        </div>
                      </div>

                      <div className="bg-stone-50 rounded-2xl p-5 border border-stone-200/85">
                        <h4 className="text-xs font-black uppercase tracking-wider text-stone-600 pb-2 border-b mb-3">Instruções legais e vigentes</h4>
                        <div className="text-[11.5px] text-stone-600 leading-relaxed font-sans flex flex-col gap-3">

                            Ao selecionar uma secretaria e um encarregado de oitiva no painel esquerdo, você poderá simular a criação de uma certidão digital de ouvidoria de controle interno, bem como exportá-lo diretamente em cópia autenticada.

                    {emailLogs.length === 0 ? (
                      <div className="text-center py-12 bg-stone-50/50 rounded-2xl border-2 border-dashed">
                        <Mail className="w-10 h-10 text-stone-300 mx-auto mb-2" />
                        <h5 className="text-sm font-bold text-stone-500">Nenhum e-mail disparado</h5>

                          As notificações são disparadas automaticamente assim que um servidor ou ouvidor altera o status governamental de qualquer chamado aberto.

                    ) : (
                      <div className="flex flex-col gap-4">
                        <div className="bg-[#EBF3FE] border border-blue-200 rounded-xl p-4 flex gap-3 text-stone-800 items-start">
                          <Bell className="w-5 h-5 text-[#1351B4] flex-shrink-0 mt-0.5" />
